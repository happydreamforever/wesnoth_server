cmake_minimum_required(VERSION 3.22)
project(wesnoth_alliance_bridge VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BOOST_INCLUDE_DIR "" CACHE PATH "Existing Boost headers")
if(NOT EXISTS "${BOOST_INCLUDE_DIR}/boost/json/src.hpp")
    message(FATAL_ERROR "BOOST_INCLUDE_DIR must contain the existing Boost JSON and Asio headers")
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(alliance-account-bridge main.cpp)
target_include_directories(alliance-account-bridge SYSTEM PRIVATE "${BOOST_INCLUDE_DIR}")
target_compile_definitions(alliance-account-bridge PRIVATE BOOST_ALL_NO_LIB BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(alliance-account-bridge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(alliance-account-bridge PRIVATE OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB Threads::Threads)
if(WIN32)
    target_compile_definitions(alliance-account-bridge PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(alliance-account-bridge PRIVATE ws2_32 mswsock)
endif()
set_target_properties(alliance-account-bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
